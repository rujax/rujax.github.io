目标

本文的目标是通过使用 Qt 的网络模块 Qt Network 从订阅链接获取返回的 XML 数据后，
再使用 Qt XML 模块将内容解析后全部打印出来。

使用 Qt Network

CMakeLists.txt

打开 CMakeLists.txt 文件，添加如下高亮代码：

...

set(CMAKE_AUTORCC ON)

find_package(Qt6 6.5 REQUIRED COMPONENTS Core Widgets Network)

qt_standard_project_setup()

...

target_include_directories(RSSReader PRIVATE ${CMAKE_CURRENT_BINARY_DIR})

target_link_libraries(RSSReader
    PRIVATE
        Qt::Core
        Qt::Widgets
        Qt::Network
)

...

mainwindow.h

打开 mainwindow.h 文件，添加如下高亮代码：

...

class QGridLayout;
class QNetworkAccessManager;
class QNetworkReply;

class MainWindow : public QMainWindow
{
    ...

    ~MainWindow() override;

private:
    QNetworkAccessManager *_manager;

    ...

private slots:
    void _subscribeButtonClicked();
    void _requestFinished(QNetworkReply *reply);
};

...

简单描述下代码：

  1. 声明 Qt 类 QNetworkAccessManager 和 QNetworkReply
  2. 声明一个 QNetworkAccessManager 的指针变量
  3. 声明一个私有槽，目标是为了和 QNetworkAccessManager 的 finished 信号连接

mainwindow.cpp

include 和 QNetworkAccessManager 实例化 

...

#include <QMessageBox>
#include <QNetworkAccessManager>
#include <QNetworkReply>

...

MainWindow::MainWindow(QWidget *parent) : QMainWindow(parent)
{
    setWindowTitle("RSSReader");
    
    _manager = new QNetworkAccessManager(this);

    ...
}

...

调整析构函数 

// 修改前
MainWindow::~MainWindow() = default;

// 修改后
MainWindow::~MainWindow()
{
    _manager->disconnect();
    _manager->deleteLater();
    _manager = nullptr;
}

调整 _connectSlots() 方法 

void MainWindow::_connectSlots()
{
    connect(_subscribeButton, &QPushButton::clicked, this, &MainWindow::_subscribeButtonClicked);
    connect(_manager, &QNetworkAccessManager::finished, this, &MainWindow::_requestFinished);
}

调整 _subscribeButtonClicked() 槽 

void MainWindow::_subscribeButtonClicked()
{
    // 修改前
    QMessageBox::information(this, "单行编辑组件", _feedEdit->text());

    // 修改后
    qDebug() << "feed link text:" << _feedEdit->text();

    QNetworkRequest request(QUrl(_feedEdit->text()));
    request.setRawHeader("Accept", "application/xml");

    _manager->get(request);
}

添加 _requestFinished() 槽的实现 

void MainWindow::_requestFinished(QNetworkReply *reply)
{
    if (reply->error() != QNetworkReply::NoError)
    {
        qDebug() << "request error:" << reply->errorString();

        reply->disconnect();
        reply->deleteLater();
        reply = nullptr;

        return;
    }

    QByteArray content = reply->readAll();

    qDebug() << "content:" << content;

    reply->disconnect();
    reply->deleteLater();
    reply = nullptr;
}

代码描述

  1. include 用到的 Qt 库 QNetworkAccessManager 和 QNetworkReply
  2. 调整析构函数，在主窗口销毁时：
    1. 调用 QObject 的 disconnect() 方法断开信号和槽的连接
    2. 调用 QObject 的 deleteLater() 槽，在合适时机删除对象
    3. 设置对象为空指针，释放内存空间
  3. 连接信号 QNetworkAccessManager::finished 和槽 MainWindow::_requestFinished
  4. 在 _subscribeButtonClicked() 槽中：
    1. 调用 QUrl 的 构造函数 构造一个 QUrl 对象
    2. 调用 QNetworkRequest 的 构造函数 构造一个 QNetworkRequest 对象 request
    3. 调用 QNetworkRequest 的 setRawHeader() 方法设置 Accept 请求头，内容为 application/xml
    4. 调用 QNetworkAccessManager 的 get() 方法发起请求
  5. 在 _requestFinished() 槽中：
    1. 判断响应 QNetworkReply 的对象指针 reply 是否存在错误，
      存在的话则调用继承自 QIODevice 的 errorString() 方法输出错误信息并将对象指针做安全删除
    2. 当 reply 不存在错误时，调用继承自 QIODevice 的 readAll() 方法读取全部返回内容

使用 Qt Xml

CMakeLists.txt

打开 CMakeLists.txt 文件，添加如下高亮代码：

...

set(CMAKE_AUTORCC ON)

find_package(Qt6 6.5 REQUIRED COMPONENTS Core Widgets Network Xml)

qt_standard_project_setup()

...

target_include_directories(RSSReader PRIVATE ${CMAKE_CURRENT_BINARY_DIR})

target_link_libraries(RSSReader
    PRIVATE
        Qt::Core
        Qt::Widgets
        Qt::Network
        Qt::Xml
)

...

mainwindow.cpp

打开 mainwindow.cpp 文件，添加如下高亮代码：

...

#include <QNetworkReply>
#include <QXmlStreamReader>

...

void MainWindow::_requestFinished(QNetworkReply *reply)
{
    ...

    QByteArray content = reply->readAll();
    QXmlStreamReader reader(content);
    QString name;
    QString indent = "  ";

    while (!reader.atEnd()) {
        reader.readNext();

        if (reader.isStartElement())
        {
            name = reader.name().toString();
            QString lowerName = name.toLower();

            if (lowerName == "item")
            {
                indent = "    ";
                qDebug() << "\n  item: {";
            }
            else if (lowerName == "channel")
            {
                indent = "  ";
                qDebug() << "\nchannel: {";
            }
            else if (lowerName == "rss")
            {
                qDebug() << "rss version:" << reader.attributes().value("version").toString();
            }
        } else if (reader.isEndElement())
        {
            QString lowerName = reader.name().toString().toLower();

            if (lowerName == "item") qDebug() << "  }";
            else if (lowerName == "channel") qDebug() << "}";
        } else if (reader.isCharacters() && !reader.isWhitespace())
        {
            QDebug debug = qDebug();
            debug.noquote() << QString("%1%2:").arg(indent, name);
            debug.quote() << reader.text().toString();
        }
    }

    if (reader.hasError())
    {
        qDebug() << "parse xml error:" << reader.errorString();
    }

    reply->disconnect();

    ...
}

简单描述下代码：

  1. include 用到的 Qt 库 QXmlStreamReader
  2. 调用 QXmlStreamReader 的 构造函数 构建对象 reader
  3. 声明 QString 对象 name 和 indent 在后续输出中使用
  4. 使用 while() 循环，判断条件为 QXmlStreamReader 的 atEnd() 方法的返回值
  5. 调用 QXmlStreamReader 的 readNext() 方法读取下一个标记
  6. 调用 QXmlStreamReader 的 isStartElement() 方法判断当前标记是否为起始元素
    1. 依次调用 name() 方法和 QStringView 的 toString() 方法获取当前元素的名称
    2. 如果名称为 item 或者 channel 的话，调整每一项的缩进并输出带有缩进的名称和 {
    3. 如果名称为 rss 的话，则依次调用 attributes() 方法和 QXmlStreamAttributes 的 value() 方法
      以及 QStringView 的 toString() 方法获取版本信息
  7. 调用 QXmlStreamReader 的 isEndElement() 方法判断当前标记是否为结束元素
    1. 使用与 6.1 相同的方式获取当前元素的名称
    2. 如果名称为 item 或者 channel 的话，输出带有缩进的 }
  8. 调用 QXmlStreamReader 的 isCharacters() 方法和 isWhitespace() 方法
    判断当前元素的文本是否是字符，同时字符是否不全是空格
    1. 调用 qDebug() 宏获得 QDebug 对象 debug
    2. 调用 noquote() 方法后输出索引和元素名称（输出的字符串不显示双引号）
    3. 调用 quote() 方法后输出元素文本（输出的字符串恢复显示双引号）
  9. 调用 QXmlStreamReader 的 hasError() 方法，如果存在错误，则调用 errorString() 方法输出错误信息
