// 设置图片
void Channel::setImageUrl(const QUrl &url)
{
    if (!url.isValid())
    {
        return;
    }

    QNetworkRequest request(url);
    QNetworkReply *reply = _manager->get(request);

    connect(reply, &QNetworkReply::finished, this, [this, reply]() { _requestFinished(reply); });
}

// 窗口大小变化时，重新设置每个组件的高度以及字体大小
void Channel::resizeEvent(QResizeEvent *event)
{
    QWidget::resizeEvent(event);

    int availableWidth = this->width() - 10 - 10;
    int availableHeight = this->height() - 10 - 10;

    if (availableWidth > 0 && availableHeight > 0)
    {
        int bannerHeight = availableWidth * 100 / 220;
        _bannerLabel->setFixedHeight(bannerHeight);

        double scaleFactor = static_cast<double>(availableWidth) / 220.0;

        int titleSize = qRound(20.0 * scaleFactor);

        if (titleSize < 20)
        {
            titleSize = 20;
        }

        QFont titleFont = this->font();
        titleFont.setBold(true);
        titleFont.setPixelSize(titleSize);
        _titleLabel->setFont(titleFont);

        int titleHeight = qRound(30.0 * scaleFactor);

        if (titleHeight < 30)
        {
            titleHeight = 30;
        }

        _titleLabel->setFixedHeight(titleHeight);

        int descriptionSize = qRound(12.0 * scaleFactor);

        if (descriptionSize < 12)
        {
            descriptionSize = 12;
        }

        QFont descFont = this->font();
        descFont.setPixelSize(descriptionSize);
        _descriptionLabel->setFont(descFont);

        int descriptionHeight = availableHeight - bannerHeight - titleHeight;

        if (descriptionHeight > 0)
        {
            _descriptionLabel->setFixedHeight(descriptionHeight);
        }
    }

    if (_titleLabel->width() > 0)
    {
        QFontMetrics metrics(_titleLabel->font());
        QString elidedText = metrics.elidedText(_title, Qt::ElideRight, _titleLabel->width());

        _titleLabel->setText(elidedText);
    }

    if (!_pixmap.isNull() && _bannerLabel->width() > 0 && _bannerLabel->height() > 0)
    {
        QPixmap Pixmap = _cropAndRoundPixmap(_pixmap);
        _bannerLabel->setPixmap(Pixmap);
    }
}

// 鼠标点击时发射 clicked 事件
void Channel::mouseReleaseEvent(QMouseEvent *event)
{
    if (event->button() == Qt::LeftButton)
    {
        if (this->rect().contains(event->pos()))
        {
            emit clicked(_link);
            event->accept();

            return;
        }
    }

    QWidget::mouseReleaseEvent(event);
}

// 构建 UI
void Channel::_buildUI()
{
    _bannerLabel = new QLabel();
    _bannerLabel->setProperty("class", "banner");
    _bannerLabel->setScaledContents(false);
    _bannerLabel->setMinimumSize(0, 0);
    _bannerLabel->setSizePolicy(QSizePolicy::Ignored, QSizePolicy::Ignored);

    _titleLabel = new QLabel();
    _titleLabel->setProperty("class", "title");
    _titleLabel->setContentsMargins(2, 0, 2, 0);
    _titleLabel->setAlignment(Qt::AlignVCenter);
    _titleLabel->setFixedHeight(30);
    _titleLabel->setToolTip(_title);
    _titleLabel->setMinimumSize(0, 0);
    _titleLabel->setSizePolicy(QSizePolicy::Ignored, QSizePolicy::Ignored);

    _descriptionLabel = new QLabel(_description);
    _descriptionLabel->setProperty("class", "description");
    _descriptionLabel->setContentsMargins(2, 0, 2, 0);
    _descriptionLabel->setAlignment(Qt::AlignTop);
    _descriptionLabel->setWordWrap(true);
    _descriptionLabel->setToolTip(_description);
    _descriptionLabel->setMinimumSize(0, 0);
    _descriptionLabel->setSizePolicy(QSizePolicy::Ignored, QSizePolicy::Ignored);

    QVBoxLayout *mainLayout = new QVBoxLayout();
    mainLayout->setSpacing(0);
    mainLayout->setContentsMargins(10, 10, 10, 10);

    mainLayout->addWidget(_bannerLabel);
    mainLayout->addWidget(_titleLabel);
    mainLayout->addWidget(_descriptionLabel);

    setLayout(mainLayout);

// 将图片裁剪大小并做圆角化处理
QPixmap Channel::_cropAndRoundPixmap(const QPixmap &pixmap)
{
    if (pixmap.isNull())
    {
        return QPixmap();
    }

    int cropWidth = _bannerLabel->width();
    int cropHeight = _bannerLabel->height();

    if (cropWidth <= 0 || cropHeight <= 0)
    {
        return pixmap;
    }

    QPixmap targetPixmap(cropWidth, cropHeight);
    targetPixmap.fill(Qt::transparent);

    QPixmap scaledPixmap = pixmap.scaled(cropWidth,
                                         cropHeight,
                                         Qt::KeepAspectRatioByExpanding,
                                         Qt::SmoothTransformation);

    int cropX = (scaledPixmap.width() - cropWidth) / 2;
    int cropY = (scaledPixmap.height() - cropHeight) / 2;

    QPixmap croppedPixmap = scaledPixmap.copy(cropX, cropY, cropWidth, cropHeight);

    QPainter painter(&targetPixmap);
    painter.setRenderHint(QPainter::Antialiasing, true);
    painter.setRenderHint(QPainter::SmoothPixmapTransform, true);

    QBrush brush(croppedPixmap);
    painter.setBrush(brush);
    painter.setPen(Qt::NoPen);
    painter.drawRoundedRect(QRectF(0.5, 0.5, cropWidth - 1.0, cropHeight - 1.0), 10, 10);

    return targetPixmap;
}

// 接收请求网络图片的返回数据
void Channel::_requestFinished(QNetworkReply *reply)
{
    reply->deleteLater();

    if (reply->error() == QNetworkReply::NoError)
    {
        QByteArray imageData = reply->readAll();

        if (_pixmap.loadFromData(imageData))
        {
            _bannerLabel->setPixmap(_cropAndRoundPixmap(_pixmap));
        }
    }
    else
    {
        qDebug() << "request banner error:" << reply->errorString();
    }
}

添加 FlowLayout 的代码

flowlayout.h 

#pragma once

#include <QLayout>
#include <QRect>
#include <QStyle>

class FlowLayout : public QLayout
{
public:
    explicit FlowLayout(int margin = -1,
                        int hSpacing = -1,
                        int vSpacing = -1,
                        QWidget *parent = nullptr);

    ~FlowLayout();

    void addItem(QLayoutItem *item) override;
    Qt::Orientations expandingDirections() const override;
    bool hasHeightForWidth() const override;
    int heightForWidth(int) const override;
    int count() const override;
    QSize minimumSize() const override;
    void setGeometry(const QRect &rect) override;
    QSize sizeHint() const override;
    QLayoutItem *itemAt(int index) const override;
    QLayoutItem *takeAt(int index) override;

    int horizontalSpacing() const;
    int verticalSpacing() const;

private:
    enum class Mode { Measure, Arrange };

    QList<QLayoutItem *> _items;

    int _hSpacing;
    int _vSpacing;

    int _execLayout(const QRect &rect, Mode mode) const;
    int _getSpacing(QStyle::PixelMetric pm) const;
};

简单描述一下代码：

  1. 定义 _hSpacing 和 _vSpacing 处理横向和纵向的间隔
  2. 定义 _items 存储布局中的每一项
  3. 将继承自 QLayout 的大量方法覆盖重写
  4. 定义 _execLayout 计算并布局每一项，_getSpacing 则获取间隔

flowlayout.cpp 

#include "FlowLayout.h"

#include <QWidget>

FlowLayout::FlowLayout(int margin, int hSpacing, int vSpacing, QWidget *parent)
    : QLayout(parent)
    , _hSpacing(hSpacing)
    , _vSpacing(vSpacing)
{
    setContentsMargins(margin, margin, margin, margin);
}

FlowLayout::~FlowLayout()
{
    qDeleteAll(_items);
    _items.clear();
}

void FlowLayout::addItem(QLayoutItem *item)
{
    _items.append(item);
}

int FlowLayout::horizontalSpacing() const
{
    if (_hSpacing >= 0)
    {
        return _hSpacing;
    }

    return _getSpacing(QStyle::PM_LayoutHorizontalSpacing);
}

int FlowLayout::verticalSpacing() const
{
    if (_vSpacing >= 0)
    {
        return _vSpacing;
    }

    return _getSpacing(QStyle::PM_LayoutVerticalSpacing);
}

int FlowLayout::count() const
{
    return _items.size();
}

QLayoutItem *FlowLayout::itemAt(int index) const
{
    return _items.value(index);
}

QLayoutItem *FlowLayout::takeAt(int index)
{
    if (index >= 0 && index < _items.size())
    {
        return _items.takeAt(index);
    }

    return nullptr;
}

Qt::Orientations FlowLayout::expandingDirections() const
{
    return {};
}

bool FlowLayout::hasHeightForWidth() const
{
    return true;
}

int FlowLayout::heightForWidth(int width) const
{
    return _execLayout(QRect(0, 0, width, 0), Mode::Measure);
}

QSize FlowLayout::minimumSize() const
{
    QSize size;

    for (const QLayoutItem *item : std::as_const(_items))
    {
        size = size.expandedTo(item->minimumSize());
    }

    const QMargins margins = contentsMargins();

    size += QSize(margins.left() + margins.right(), margins.top() + margins.bottom());

    return size;
}

void FlowLayout::setGeometry(const QRect &rect)
{
    QLayout::setGeometry(rect);

    _execLayout(rect, Mode::Arrange);
}

QSize FlowLayout::sizeHint() const
{
    return minimumSize();
}

int FlowLayout::_execLayout(const QRect &rect, Mode mode) const
{
    int left, top, right, bottom;
    getContentsMargins(&left, &top, &right, &bottom);

    QRect effectiveRect = rect.adjusted(+left, +top, -right, -bottom);
    int availableWidth = effectiveRect.width();

    int hSpacing = horizontalSpacing();

    if (hSpacing < 0)
    {
        hSpacing = 0;
    }

    int vSpacing = verticalSpacing();

    if (vSpacing < 0)
    {
        vSpacing = 0;
    }

    int maxItemsInRow = 3;
    int calculatedWidth = (availableWidth - (hSpacing * (maxItemsInRow - 1))) / maxItemsInRow;

    if (calculatedWidth < 1)
    {
        calculatedWidth = 1;
    }

    int calculatedHeight = calculatedWidth * 180 / 240;
    int x = effectiveRect.x();
    int y = effectiveRect.y();
    int currentItemCount = 0;

    for (int i = 0; i < _items.size(); ++i)
    {
        QLayoutItem *item = _items.at(i);

        if (currentItemCount >= maxItemsInRow && i > 0)
        {
            x = effectiveRect.x();
            y = y + calculatedHeight + vSpacing;
            currentItemCount = 0;
        }

        if (mode == Mode::Arrange)
        {
            item->setGeometry(QRect(x, y, calculatedWidth, calculatedHeight));
        }

        x += calculatedWidth + hSpacing;
        currentItemCount++;
    }

    if (_items.isEmpty())
    {
        return bottom;
    }

    int totalRows = (_items.size() + maxItemsInRow - 1) / maxItemsInRow;

    int totalLayoutHeight = (totalRows * calculatedHeight) + ((totalRows - 1) * vSpacing) + top
                            + bottom;

    return totalLayoutHeight;
}

int FlowLayout::_getSpacing(QStyle::PixelMetric pm) const
{
    QObject *p = parent();

    if (!p)
    {
        return -1;
    }

    if (p->isWidgetType())
    {
        QWidget *pw = static_cast<QWidget *>(p);

        return pw->style()->pixelMetric(pm, nullptr, pw);
    }

    return static_cast<QLayout *>(p)->spacing();
}

void MainWindow::_requestFinished(QNetworkReply *reply)
{
    _subscribeButton->setEnabled(true);
    reply->deleteLater();

    if (reply->error() != QNetworkReply::NoError)
    {
        qDebug() << "request error:" << reply->errorString();

        return;
    }

    QByteArray content = reply->readAll();
    QXmlStreamReader reader(content);

    // 将 xml 返回的数据组装为 channel
    Channel *channel = new Channel(&_manager);
    Item currentItem;
    bool inItem = false;

    while (!reader.atEnd())
    {
        reader.readNext();

        if (reader.isStartElement())
        {
            QString name = reader.name().toString().toLower();

            if (name == "item")
            {
                inItem = true;
                currentItem = Item();
            }
            else if (name == "title")
            {
                QString title = reader.readElementText();

                if (inItem)
                {
                    currentItem.setTitle(title);
                }
                else
                {
                    channel->setTitle(title);
                }
            }
            else if (name == "description")
            {
                QString description = reader.readElementText();

                if (inItem)
                {
                    currentItem.setDescription(description);
                }
                else
                {
                    channel->setDescription(description);
                }
            }
            else if (name == "link")
            {
                QString uri = reader.namespaceUri().toString();

                if (uri.isEmpty())
                {
                    QString linkText = reader.readElementText();

                    if (!linkText.isEmpty())
                    {
                        QUrl link(linkText);

                        if (inItem)
                        {
                            currentItem.setLink(link);
                        }
                        else
                        {
                            channel->setLink(link);
                        }
                    }
                }
            }
            else if (name == "pubdate")
            {
                if (inItem)
                {
                    QDateTime pubDate = Item::parsePubDate(reader.readElementText());

                    if (pubDate.isValid())
                    {
                        currentItem.setPubDate(pubDate);
                    }
                }
            }
        }
        else if (reader.isEndElement())
        {
            QString name = reader.name().toString().toLower();

            if (name == "item")
            {
                inItem = false;

                channel->addItem(currentItem);
                currentItem = Item();
            }
        }
    }

    if (reader.hasError())
    {
        delete channel;

        qDebug() << "parse xml error:" << reader.errorString();

        return;
    }

    channel->setImageUrl(QUrl("https://picsum.photos/660/300.webp"));
    channel->setProperty("class", "channel");

    connect(channel, &Channel::clicked, this, [](const QUrl &link) { qDebug() << "link:" << link; });

    _channelsLayout->addWidget(channel);
}
